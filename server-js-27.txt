// run fn inside one DB transaction (all-or-nothing)
async function tx(fn) {
  const c = await pool.connect();
  try { await c.query('BEGIN'); const r = await fn(c); await c.query('COMMIT'); return r; }
  catch (e) { await c.query('ROLLBACK'); throw e; }
  finally { c.release(); }
}
const wrap = f => (req, res) => f(req, res).catch(e => res.status(400).json({ error: e.message }));
const money = v => { const n = Number(v); if (!(n > 0) || n > 1e9) throw new Error('Invalid amount'); return n.toFixed(6); };

function auth(role) {
  return (req, res, next) => {
    try {
      const u = jwt.verify((req.headers.authorization || '').replace('Bearer ', ''), SECRET);
      if (role && u.role !== role) return res.status(403).json({ error: 'Not allowed' });
      req.user = u; next();
    } catch { res.status(401).json({ error: 'Please log in' }); }
  };
}
const token = u => jwt.sign({ id: u.id, role: u.role }, SECRET, { expiresIn: '7d' });

  const r = await c.query('SELECT id FROM users WHERE referral_code = $1', [String(code || '').trim().toUpperCase()]);

}
app.post('/api/register', wrap(async (req, res) => {
  const { email, password, ref } = req.body;
  if (!email || !password || password.length < 8) throw new Error('Email and a password of 8+ characters required');
  const u = await tx(async c => {
    let referrerId = null;
    if (ref) { referrerId = await findByCode(c, ref); if (!referrerId) throw new Error('Invalid referral code'); }
    const r = await c.query('INSERT INTO users(email,password_hash,referred_by) VALUES($1,$2,$3) RETURNING id,role',
      [email.toLowerCase(), await bcrypt.hash(password, 10), referrerId]);
    const newId = r.rows[0].id;
    await c.query('UPDATE users SET referral_code=$1 WHERE id=$2', ['MRKD' + newId, newId]); // starting code; user can change it any time
    await c.query('INSERT INTO wallets(user_id) VALUES($1)', [newId]);
    return r.rows[0];
  });
  res.json({ token: token(u) });
}));

// ---------- Copy-trade engine ----------
// Runs the moment a new signal is published: buys `per_trade_usdt` worth
// of the pair for every user who has Copy ON. One user's failure (e.g. low
// balance) never stops the others - each is tried and logged independently.
async function copySignalToUsers(signal) {
  const users = await pool.query(
    'SELECT user_id, api_key_enc, api_secret_enc, per_trade_usdt FROM binance_links WHERE copy_enabled=true'
  );
  for (const u of users.rows) {
    try {
      const apiKey = binance.decrypt(u.api_key_enc), apiSecret = binance.decrypt(u.api_secret_enc);
      const order = await binance.marketBuy(apiKey, apiSecret, signal.pair, Number(u.per_trade_usdt));
      const invested = order.cummulativeQuoteQty || u.per_trade_usdt; // actual USDT spent, from Binance's own order response
      await pool.query(
        `INSERT INTO copied_trades(signal_id,user_id,binance_order_id,qty,invested_usdt,status)
         VALUES ($1,$2,$3,$4,$5,'open')`,
        [signal.id, u.user_id, String(order.orderId), order.executedQty || null, invested]
      );
    } catch (e) {
      await pool.query(
        `INSERT INTO copied_trades(signal_id,user_id,status) VALUES ($1,$2,'failed')`,
        [signal.id, u.user_id]
      );
      console.log('Copy failed for user', u.user_id, '-', e.message); // visible in Render Logs
    }
  }
}

app.post('/api/admin/withdraw/:id/:action', auth('admin'), wrap(async (req, res) => {
  const paid = req.params.action === 'paid';
  if (!paid && req.params.action !== 'reject') throw new Error('Unknown action');
  await tx(async c => {
    const w = (await c.query("SELECT * FROM withdrawals WHERE id=$1 AND status='pending' FOR UPDATE", [req.params.id])).rows[0];
    if (!w) throw new Error('Withdrawal not found or already handled');
    await c.query('UPDATE withdrawals SET status=$1, txid=$2 WHERE id=$3', [paid ? 'paid' : 'rejected', req.body.txid || null, w.id]);
    if (paid) {
      await c.query('UPDATE wallets SET locked=locked-$1 WHERE user_id=$2', [w.amount, w.user_id]);
      await c.query("INSERT INTO ledger(user_id,kind,amount,ref_id,note) VALUES($1,'withdraw',$2,$3,$4)", [w.user_id, -w.amount, w.id, w.address]);
    } else {
      await c.query('UPDATE wallets SET locked=locked-$1, balance=balance+$1 WHERE user_id=$2', [w.amount, w.user_id]);
    }
  });
  res.json({ ok: true });
}));

// ---------- Fee charge (used by the trade engine in Step 2) ----------
// Only on profit. Moves fee from user wallet to admin wallet in one transaction.
async function chargeFee(c, userId, profit, tradeId) {
  if (!(profit > 0)) return 0;
  // Referred users (someone shared their link with them) get 20% off the profit-share fee, as a signup benefit.
  const u = await c.query('SELECT referred_by FROM users WHERE id=$1', [userId]);
  const rate = u.rows[0]?.referred_by ? FEE * 0.8 : FEE;
  const fee = (profit * rate).toFixed(6);
  const admin = (await c.query("SELECT id FROM users WHERE role='admin' LIMIT 1")).rows[0];
  const w = (await c.query('SELECT balance FROM wallets WHERE user_id=$1 FOR UPDATE', [userId])).rows[0];
  if (Number(w.balance) < Number(fee)) throw new Error('Wallet too low: pause copying');
  await c.query('UPDATE wallets SET balance=balance-$1 WHERE user_id=$2', [fee, userId]);
  await c.query('UPDATE wallets SET balance=balance+$1 WHERE user_id=$2', [fee, admin.id]);
  await c.query("INSERT INTO ledger(user_id,kind,amount,ref_id,note) VALUES($1,'fee_charge',$2,$3,'profit share')", [userId, -fee, tradeId]);
  await c.query("INSERT INTO ledger(user_id,kind,amount,ref_id,note) VALUES($1,'fee_income',$2,$3,'profit share')", [admin.id, fee, tradeId]);
  return fee;
}

// ---------- Subscription fee: a fixed number of MRKD tokens every N days ----------
// Set SUBSCRIPTION_FEE_AMOUNT in Render to switch this on (0 or unset = disabled).
// This never runs the user's balance negative: if they can't afford it, it's skipped
// and tried again next check, rather than ever going below zero.
async function runSubscriptionFees() {
  const amount = Number(process.env.SUBSCRIPTION_FEE_AMOUNT || 0);
  const days = Number(process.env.SUBSCRIPTION_FEE_DAYS || 15);
  if (!(amount > 0)) return;
  const admin = (await pool.query("SELECT id FROM users WHERE role='admin' LIMIT 1")).rows[0];
  if (!admin) return;
  const due = await pool.query(
    `SELECT user_id, balance FROM wallets WHERE last_fee_at < now() - ($1 || ' days')::interval AND user_id<>$2`,
    [days, admin.id]
  );
  for (const w of due.rows) {
    await tx(async c => {
      const row = (await c.query('SELECT balance FROM wallets WHERE user_id=$1 FOR UPDATE', [w.user_id])).rows[0];
      if (Number(row.balance) < amount) { // can't afford it yet - just push the due date so we don't spam-check every hour
        await c.query('UPDATE wallets SET last_fee_at=now() WHERE user_id=$1', [w.user_id]);
        return;
      }
      await c.query('UPDATE wallets SET balance=balance-$1, last_fee_at=now() WHERE user_id=$2', [amount, w.user_id]);
      await c.query('UPDATE wallets SET balance=balance+$1 WHERE user_id=$2', [amount, admin.id]);
      await c.query("INSERT INTO ledger(user_id,kind,amount,note) VALUES($1,'fee_charge',$2,'subscription fee')", [w.user_id, -amount]);
      await c.query("INSERT INTO ledger(user_id,kind,amount,note) VALUES($1,'fee_income',$2,'subscription fee')", [admin.id, amount]);
    }).catch(e => console.log('Subscription fee error for user', w.user_id, '-', e.message));
  }
}

  if (LINK_PATTERN.test(text)) throw new Error('Links are not allowed in the group chat');
  const lower = text.toLowerCase();
  if (CHAT_BANNED.some(w => lower.includes(w))) throw new Error('Please keep the chat respectful - that word is not allowed');
  const r = await pool.query('INSERT INTO chat_messages(user_id, text) VALUES ($1,$2) RETURNING id, created_at', [req.user.id, text]);
  res.json({ ok: true, id: r.rows[0].id, created_at: r.rows[0].created_at });
}));
